### Phase 1: マルチシグナル・ヒットスコアの設計

#### 1-1. 現状のスコア計算の問題点

現在（ranking_service.py:139-146）:
```python
hit_score = (total_spend_increase / max_spend) * 100 * (1 / rank ** 0.3)
is_hit = rank <= 20 and total_spend_increase > spend_threshold and total_spend_increase > 0
```

問題:
- spend_increase が推定値 → 不正確
- 単一指標（消化額）のみ → 多角的でない
- rank^0.3 の調整が恣意的
- 配信日数や生存情報を考慮していない

#### 1-2. 新しいスコア計算モデル

**ファイル修正**: `backend/app/services/ranking/ranking_service.py`

新しいヒットスコア計算:

```python
def _compute_multi_signal_hit_score(
    ad: Ad,
    metrics: list[AdDailyMetrics],
    genre_stats: dict,  # ジャンル内の統計
) -> tuple[float, bool, dict]:
    """
    マルチシグナル・ヒットスコアを計算。

    Returns:
        (hit_score: 0-100, is_hit: bool, score_breakdown: dict)
    """
    signals = {}

    # ── Signal 1: 配信継続力 (0-25点) ──
    # 長く配信されている = 広告主がROIを感じている証拠
    meta = ad.ad_metadata or {}
    days_running = 0
    if ad.first_seen_at:
        end = ad.last_seen_at or datetime.now(timezone.utc)
        days_running = max(1, (end - ad.first_seen_at).days)

    is_still_running = meta.get("is_still_running", ad.last_seen_at is None)

    # 30日以上配信 = 5点, 60日以上 = 15点, 90日以上 = 25点
    if days_running >= 90:
        longevity = 25
    elif days_running >= 60:
        longevity = 15 + (days_running - 60) / 30 * 10
    elif days_running >= 30:
        longevity = 5 + (days_running - 30) / 30 * 10
    else:
        longevity = days_running / 30 * 5

    # まだ配信中ならボーナス +5
    if is_still_running and days_running >= 14:
        longevity = min(25, longevity + 5)

    signals["longevity"] = round(longevity, 1)

    # ── Signal 2: 消化額の相対位置 (0-25点) ──
    total_spend = sum(m.estimated_spend_increase or 0 for m in metrics)
    genre_max_spend = genre_stats.get("max_spend", 1)
    genre_p50_spend = genre_stats.get("p50_spend", 0)

    if genre_max_spend > 0 and total_spend > 0:
        # ジャンル内パーセンタイル
        spend_ratio = total_spend / genre_max_spend
        spend_score = min(25, spend_ratio * 25)

        # P50超えならボーナス
        if total_spend > genre_p50_spend * 2:
            spend_score = min(25, spend_score + 5)
    else:
        spend_score = 0

    signals["spend"] = round(spend_score, 1)

    # ── Signal 3: オーディエンスサイズ (0-20点) ──
    audience_min = meta.get("estimated_audience_min", 0) or 0
    audience_max = meta.get("estimated_audience_max", 0) or 0
    avg_audience = (audience_min + audience_max) / 2

    genre_max_audience = genre_stats.get("max_audience", 1)
    if avg_audience > 0 and genre_max_audience > 0:
        audience_score = min(20, (avg_audience / genre_max_audience) * 20)
    else:
        audience_score = 0

    signals["audience"] = round(audience_score, 1)

    # ── Signal 4: 配信面の広さ (0-10点) ──
    platforms = meta.get("publisher_platforms", [])
    platform_score = min(10, len(platforms) * 3)  # 3面以上で満点
    signals["platforms"] = round(platform_score, 1)

    # ── Signal 5: クリエイティブ品質 (0-10点) ──
    creative_score = 0
    if ad.video_url:
        creative_score += 4  # 動画あり
    if ad.image_url:
        creative_score += 3  # 画像あり
    if ad.thumbnail_url and "s200x200" not in (ad.thumbnail_url or "") and "favicons" not in (ad.thumbnail_url or ""):
        creative_score += 3  # 高品質サムネイル
    signals["creative"] = min(10, creative_score)

    # ── Signal 6: トレンド加速度 (0-10点) ──
    if len(metrics) >= 3:
        recent = metrics[-3:]  # 直近3日
        older = metrics[:-3] if len(metrics) > 3 else metrics[:1]

        recent_avg = sum(m.view_count_increase or 0 for m in recent) / len(recent)
        older_avg = sum(m.view_count_increase or 0 for m in older) / max(len(older), 1)

        if older_avg > 0:
            acceleration = (recent_avg - older_avg) / older_avg
            trend_score = min(10, max(0, acceleration * 20))
        else:
            trend_score = 5 if recent_avg > 0 else 0
    else:
        trend_score = 0

    signals["trend"] = round(trend_score, 1)

    # ── 合計 ──
    hit_score = sum(signals.values())
    hit_score = min(100, max(0, round(hit_score, 1)))

    # ヒット判定: 60点以上 かつ 配信14日以上
    is_hit = hit_score >= 60 and days_running >= 14

    return hit_score, is_hit, signals
```

### Phase 2: ジャンル別統計の事前計算

#### 2-1. ジャンル統計の算出
**ranking_service.py の _compute_rankings() メソッド内に追加**:

```python
def _compute_genre_stats(self, session, period_start, period_end) -> dict:
    """ジャンル別の統計を事前計算"""
    from sqlalchemy import func

    genre_stats = {}

    # 全広告のメトリクスをジャンル別に集計
    results = session.query(
        Ad.genre,
        func.sum(AdDailyMetrics.estimated_spend_increase).label("total_spend"),
        func.max(AdDailyMetrics.estimated_spend_increase).label("max_spend"),
        func.count(Ad.id).label("ad_count"),
    ).join(
        AdDailyMetrics, Ad.id == AdDailyMetrics.ad_id
    ).filter(
        AdDailyMetrics.date >= period_start,
        AdDailyMetrics.date <= period_end,
    ).group_by(Ad.genre).all()

    for row in results:
        genre = row.genre or "other"
        genre_stats[genre] = {
            "total_spend": row.total_spend or 0,
            "max_spend": row.max_spend or 0,
            "ad_count": row.ad_count or 0,
            "p50_spend": (row.total_spend or 0) / max(row.ad_count or 1, 1),
            "max_audience": 0,  # Agent Aのデータ取得後に更新
        }

    # オーディエンスサイズの統計を追加
    ads_with_audience = session.query(Ad).filter(
        Ad.ad_metadata.isnot(None)
    ).all()

    for ad in ads_with_audience:
        meta = ad.ad_metadata or {}
        genre = ad.genre or "other"
        audience_max = meta.get("estimated_audience_max", 0) or 0
        if genre in genre_stats:
            genre_stats[genre]["max_audience"] = max(
                genre_stats[genre]["max_audience"], audience_max
            )

    return genre_stats
```

### Phase 3: ヒット判定の閾値チューニング

#### 3-1. 動的閾値の導入

固定閾値（rank ≤ 20）ではなく、データ分布に基づく動的閾値:

```python
def _compute_dynamic_thresholds(self, scores: list[float]) -> dict:
    """スコア分布から動的にヒット閾値を決定"""
    import numpy as np

    if not scores:
        return {"hit_threshold": 60, "strong_hit_threshold": 80}

    arr = np.array(scores)
    p75 = float(np.percentile(arr, 75))
    p90 = float(np.percentile(arr, 90))

    return {
        "hit_threshold": max(50, min(70, p75)),       # 上位25%以上
        "strong_hit_threshold": max(70, min(90, p90)), # 上位10%以上
    }
```

2段階のヒット判定:
