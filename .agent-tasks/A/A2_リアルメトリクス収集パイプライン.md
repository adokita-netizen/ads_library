### Phase 1: Meta Graph API からの実メトリクス取得

#### 1-1. Meta Ad Library API のリーチ・インプレッションデータ取得
```python
# Meta Ad Library API v21.0 では ad_delivery_start_time, estimated_audience_size が取得可能
# ただし spend/impressions は政治広告のみ
#
# 取得可能なフィールド:
# - ad_delivery_start_time (開始日)
# - ad_delivery_stop_time (終了日 = まだ配信中ならNone)
# - estimated_audience_size (min/max)
# - publisher_platforms (配信面)
# - demographic_distribution (年齢・性別分布) ← 政治広告のみ
# - delivery_by_region (地域分布) ← 政治広告のみ
#
# 商用広告で取得できる最大限の情報を活用する
```

**ファイル作成**: `backend/scripts/collect_real_metrics.py`

```python
"""
Meta Ad Library API から広告の実配信データを収集する。
商用広告でも取得可能なフィールド:
  - estimated_audience_size (推定オーディエンスサイズ)
  - publisher_platforms (配信プラットフォーム)
  - ad_delivery_start_time / stop_time (配信期間)

これらを使ってより正確なメトリクス推定を行う。
"""
import sys; sys.path.insert(0, ".")
import time
import httpx
from datetime import datetime, timezone
from sqlalchemy.orm.attributes import flag_modified
from app.core.database import SyncSessionLocal
from app.models.ad import Ad
from app.models.ad_metrics import AdDailyMetrics
from app.api.endpoints.settings import load_api_keys_from_db

GRAPH_API_VERSION = "v21.0"
BATCH_SIZE = 50  # Graph API batch limit
REQUEST_DELAY = 2.0

def get_access_token():
    keys = load_api_keys_from_db()
    meta_keys = keys.get("meta", keys.get("facebook", {}))
    token = meta_keys.get("access_token")
    if not token:
        raise ValueError("Meta access_token が設定されていません。Settings画面で設定してください。")
    return token

def fetch_ad_library_data(external_ids: list[str], token: str) -> dict:
    """Meta Ad Library API でバッチ取得"""
    url = f"https://graph.facebook.com/{GRAPH_API_VERSION}/ads_archive"
    results = {}

    for ext_id in external_ids:
        try:
            params = {
                "access_token": token,
                "search_terms": "",
                "ad_reached_countries": '["JP"]',
                "fields": ",".join([
                    "id",
                    "ad_delivery_start_time",
                    "ad_delivery_stop_time",
                    "estimated_audience_size",
                    "publisher_platforms",
                    "page_name",
                    "ad_snapshot_url",
                    "ad_creative_bodies",
                    "ad_creative_link_captions",
                    "ad_creative_link_titles",
                ]),
                "search_page_ids": ext_id.split("_")[0] if "_" in ext_id else "",
                "limit": 5,
            }
            resp = httpx.get(url, params=params, timeout=15.0)
            if resp.status_code == 200:
                data = resp.json().get("data", [])
                for item in data:
                    results[item.get("id", "")] = item
            time.sleep(REQUEST_DELAY)
        except Exception as e:
            print(f"  API error for {ext_id}: {e}")

    return results

def update_ad_with_real_data(ad: Ad, api_data: dict, session) -> bool:
    """APIデータでAdレコードを更新"""
    updated = False
    meta = dict(ad.ad_metadata or {})

    # 配信期間の更新
    start_time = api_data.get("ad_delivery_start_time")
    stop_time = api_data.get("ad_delivery_stop_time")

    if start_time:
        meta["delivery_start_time"] = start_time
        # first_seen_at が未設定なら更新
        if not ad.first_seen_at:
            ad.first_seen_at = datetime.fromisoformat(start_time.replace("Z", "+00:00"))
        updated = True

    if stop_time:
        meta["delivery_stop_time"] = stop_time
        meta["is_still_running"] = False
    else:
        meta["is_still_running"] = True

    # オーディエンスサイズ
    audience = api_data.get("estimated_audience_size", {})
    if audience:
        meta["estimated_audience_min"] = audience.get("lower_bound")
        meta["estimated_audience_max"] = audience.get("upper_bound")
        updated = True

    # 配信プラットフォーム
    platforms = api_data.get("publisher_platforms", [])
    if platforms:
        meta["publisher_platforms"] = platforms
        updated = True

    # メトリクスの再計算（実データベース）
    if audience and start_time:
        _recalculate_metrics_with_real_data(ad, meta)
        updated = True

    if updated:
        ad.ad_metadata = meta
        flag_modified(ad, "ad_metadata")

    return updated

def _recalculate_metrics_with_real_data(ad: Ad, meta: dict):
    """実データからメトリクスを再計算"""
    audience_min = meta.get("estimated_audience_min", 0) or 0
    audience_max = meta.get("estimated_audience_max", 0) or 0
    avg_audience = (audience_min + audience_max) / 2 if audience_max > 0 else 0

    # 配信日数
    start_str = meta.get("delivery_start_time")
    stop_str = meta.get("delivery_stop_time")
    if start_str:
        start = datetime.fromisoformat(start_str.replace("Z", "+00:00"))
        end = datetime.fromisoformat(stop_str.replace("Z", "+00:00")) if stop_str else datetime.now(timezone.utc)
        days_running = max(1, (end - start).days)
    else:
        days_running = 1

    # 改善された推定: オーディエンスサイズ × 配信日数 × プラットフォーム数
    platforms = meta.get("publisher_platforms", [])
    platform_multiplier = max(1.0, len(platforms) * 0.8)

    # オーディエンスサイズから推定インプレッション
    # 一般的に: impressions ≈ audience_size × frequency (2-5回/人)
    estimated_frequency = 3.0
    if days_running > 90:
        estimated_frequency = 4.5
    elif days_running > 30:
        estimated_frequency = 3.5

    if avg_audience > 0:
        impressions = int(avg_audience * estimated_frequency * platform_multiplier)
        meta["impressions_from_audience"] = impressions
        meta["estimation_method"] = "audience_based"  # vs "cpm_based"
        meta["estimation_confidence"] = 0.65  # より高い信頼度

        ad.impressions = impressions
        ad.view_count = impressions

        # Spend推定もオーディエンスベースに改善
        cpm_jpy = 800
        genre = ad.genre or "other"
        genre_multipliers = {
            "beauty": 1.3, "health": 1.2, "ec_d2c": 1.1,
            "finance": 1.5, "app": 0.9, "food": 1.0,
            "education": 1.1, "technology": 1.2, "other": 1.0,
        }
        cpm = cpm_jpy * genre_multipliers.get(genre, 1.0)
        ad.spend = round(impressions * cpm / 1000)
        meta["estimated_spend_jpy"] = ad.spend
        meta["estimated_cpm_jpy"] = cpm
```

5. 結果として video_url=NULL は「動画広告ではない」ことを意味するようにする

### Phase 4: ヒットスコアの再計算

#### 4-1. ranking_service.py の改善

現在のヒットスコア計算:
```python
hit_score = (total_spend_increase / max_spend) * 100 * (1 / rank ** 0.3)
```

改善版（実データを反映）:
```python
hit_score = weighted_sum(
    spend_component     * 0.25,  # 消化額（推定 or 実）
    audience_component  * 0.20,  # オーディエンスサイズ（実データ）
    longevity_component * 0.20,  # 配信継続日数（実データ）
    platform_component  * 0.15,  # 配信プラットフォーム数（実データ）
    engagement_component* 0.10,  # いいね・シェア
    creative_quality    * 0.10,  # クリエイティブ品質スコア
)
```

各コンポーネントの計算:
- `spend_component`: ジャンル内での相対消化額 (0-100)
- `audience_component`: オーディエンスサイズのパーセンタイル (0-100)
- `longevity_component`: `min(100, days_running / 90 * 100)` — 90日以上で満点
- `platform_component`: `min(100, len(platforms) / 4 * 100)` — 4面以上で満点
- `engagement_component`: いいね数のパーセンタイル (0-100)
- `creative_quality`: メディア充実度 (0-100)
